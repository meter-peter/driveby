\begin{tikzpicture}[
    >={Stealth[length=2.5mm]},
    node distance=0.7cm,
    xrd/.style={
        rectangle,
        rounded corners=5pt,

        fill=violet!15,
        minimum width=5cm,
        minimum height=1cm,
        font=\small\bfseries,
        align=center
    },
    resource/.style={
        rectangle,
        rounded corners=3pt,
        draw=violet!50!black,
        fill=violet!8,
        minimum width=3.2cm,
        minimum height=0.6cm,
        font=\scriptsize,
        align=center
    },
    group/.style={
        rectangle,
        rounded corners=4pt,
        draw=#1!30,
        fill=#1!3,
        inner sep=8pt
    },
    grouplabel/.style={
        font=\scriptsize\bfseries,
        text=#1!70!black
    }
]

% XSDLC CR at top
\node[xrd] (xsdlc) at (3.25,0) {XSDLC\\(single CR per application)};

% --- App-level resources (left column) ---
% Group label on the left of the column so the XSDLC arrow has a clear path.
\node[grouplabel=teal,anchor=east] at (-2.0,-1.2) {App-level (shared)};

\node[resource] (sa)       at (0,-2.0) {ServiceAccount};
\node[resource] (eb)       at (0,-2.8) {EventBus};
\node[resource] (scm)      at (0,-3.6) {ScmProvider};
\node[resource] (gitrepo)  at (0,-4.4) {GitRepository};
\node[resource] (promstrat) at (0,-5.2) {PromotionStrategy};
\node[resource] (commitst) at (0,-6.0) {ArgoCDCommitStatus};

\begin{scope}[on background layer]
    \node[group=teal, fit={($(sa.north west)+(-0.3,0.3)$) ($(commitst.south east)+(0.3,-0.3)$)}] (appgrp) {};
\end{scope}

% Arrows from XSDLC to app-level
\draw[->,thick,violet!50!black] (xsdlc.south) -- ++(0,-0.5) -| (sa.north);

\draw[->,thick,teal!50!black] (sa) -- (eb);
\draw[->,thick,teal!50!black] (eb) -- (scm);
\draw[->,thick,teal!50!black] (scm) -- (gitrepo);
\draw[->,thick,teal!50!black] (gitrepo) -- (promstrat);
\draw[->,thick,teal!50!black] (promstrat) -- (commitst);

% --- Per-gate resources (right column) ---
% Group label on the right of the column so the XSDLC arrow has a clear path.
\node[grouplabel=orange,anchor=west] at (8.5,-1.2) {Per-gate (one set each)};

\node[resource] (role)    at (6.5,-2.0) {Role};
\node[resource] (rb)      at (6.5,-2.8) {RoleBinding};
\node[resource] (wft)     at (6.5,-3.6) {WorkflowTemplate};
\node[resource] (es)      at (6.5,-4.4) {EventSource};
\node[resource] (sensor)  at (6.5,-5.2) {Sensor};
\node[resource] (svc)     at (6.5,-6.0) {Service};
\node[resource] (ing)     at (6.5,-6.8) {Ingress};

\begin{scope}[on background layer]
    \node[group=orange, fit={($(role.north west)+(-0.3,0.3)$) ($(ing.south east)+(0.3,-0.3)$)}] (gategrp) {};
\end{scope}

% Arrows from XSDLC to per-gate
\draw[->,thick,violet!50!black] (xsdlc.south) -- ++(0,-0.5) -| (role.north);

\draw[->,thick,orange!50!black] (role) -- (rb);
\draw[->,thick,orange!50!black] (rb) -- (wft);
\draw[->,thick,orange!50!black] (wft) -- (es);
\draw[->,thick,orange!50!black] (es) -- (sensor);
\draw[->,thick,orange!50!black] (sensor) -- (svc);
\draw[->,thick,orange!50!black] (svc) -- (ing);

% Legend
\node[font=\tiny\itshape,text=teal!60!black] at (0,-6.9) {(one instance per XSDLC)};
\node[font=\tiny\itshape,text=orange!60!black] at (6.5,-7.7) {(replicated per gated environment)};
