% Staging Promotion Workflow DAG — Chapter 5
% Shows: set-pending-status → wait-for-source-ready → check-0-validate-only → check-1-functional-test → report-success + comment-pr + update-commitstatus-success (parallel)
% Exit handler on failure (dashed)
\begin{tikzpicture}[
    >={Stealth[length=2.5mm]},
    node distance=1.1cm,
    dagstep/.style={
        rectangle,
        rounded corners=3pt,
        draw=teal!60!black,
        fill=teal!10,
        minimum width=3.8cm,
        minimum height=0.7cm,
        font=\small,
        align=center
    },
    dagparallel/.style={
        rectangle,
        rounded corners=3pt,
        draw=green!50!black,
        fill=green!10,
        minimum width=3.2cm,
        minimum height=0.7cm,
        font=\small,
        align=center
    },
    dagfail/.style={
        rectangle,
        rounded corners=3pt,
        draw=red!60!black,
        fill=red!10,
        minimum width=3.2cm,
        minimum height=0.7cm,
        font=\small,
        align=center
    },
    dagnumber/.style={
        circle,
        draw=teal!50!black,
        fill=teal!20,
        minimum size=0.5cm,
        font=\tiny\bfseries,
        inner sep=0pt
    }
]

% Sequential steps (top to bottom)
\node[dagstep] (pending)    at (0,0)     {set-pending-status};
\node[dagstep] (health)     at (0,-1.3)  {wait-for-source-ready};
\node[dagstep] (check0)     at (0,-2.6)  {check-0-validate-only};
\node[dagstep] (check1)     at (0,-3.9)  {check-1-functional-test};

% Parallel fork (three success steps)
\node[dagparallel] (report)  at (-3.2,-5.5) {report-success};
\node[dagparallel] (comment) at (0,-5.5)    {comment-pr};
\node[dagparallel] (commitstatus) at (3.2,-5.5) {update-commitstatus\\-success};

% Step numbers
\node[dagnumber] at ($(pending.west)+(-0.5,0)$)   {1};
\node[dagnumber] at ($(health.west)+(-0.5,0)$)    {2};
\node[dagnumber] at ($(check0.west)+(-0.5,0)$)    {3};
\node[dagnumber] at ($(check1.west)+(-0.5,0)$)    {4};
\node[dagnumber] at ($(report.west)+(-0.4,0)$)    {5};
\node[dagnumber] at ($(comment.south)+(0,-0.35)$)  {6};
\node[dagnumber] at ($(commitstatus.east)+(0.4,0)$) {7};

% Sequential arrows
\draw[->,thick] (pending)  -- (health);
\draw[->,thick] (health)   -- (check0);
\draw[->,thick] (check0)   -- (check1);

% Fork arrows
\draw[->,thick] (check1.south) -- ++(0,-0.5) -| (report.north);
\draw[->,thick] (check1.south) -- ++(0,-0.5) -- (comment.north);
\draw[->,thick] (check1.south) -- ++(0,-0.5) -| (commitstatus.north);

% Fork label
\node[font=\tiny\itshape,teal!60!black] at (0,-4.65) {parallel};

% Exit handler (failure path)
\node[dagfail] (exithandler) at (5.5,-2.6) {exit-handler};
\node[dagfail,font=\scriptsize] (failgithub) at (5.5,-3.6) {report-failure-github};
\node[dagfail,font=\scriptsize] (failstatus) at (5.5,-4.4) {update-commitstatus-failure};
\node[dagfail,font=\scriptsize] (failcomment) at (5.5,-5.2) {comment-pr-failure};

\draw[->,dashed,red!60!black,thick] (check0.east) -- node[above,font=\tiny,red!60!black] {on failure} (exithandler.west);
\draw[->,dashed,red!60!black,thick] (check1.east) -- ++(1.2,0) |- (exithandler.south west);

% Exit handler parallel steps
\draw[->,thick,red!40!black] (exithandler) -- (failgithub);
\draw[->,thick,red!40!black] (failgithub) -- (failstatus);
\draw[->,thick,red!40!black] (failstatus) -- (failcomment);
\node[font=\tiny\itshape,red!60!black] at (6.8,-4.0) {parallel};

\end{tikzpicture}
