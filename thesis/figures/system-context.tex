% System Context Diagram — System Architecture chapter
% Shows: the system under study (DriveBy CLI + XSDLC composition + generated

% encloses every component that participates in producing the promotion verdict.
\begin{tikzpicture}[

        align=center

    actor/.style={
        rectangle,
        rounded corners=2pt,
        draw=#1!60!black,
        fill=#1!10,

        minimum height=1.0cm,
        font=\scriptsize\bfseries,
        align=center
    },
    flow/.style={
        ->,
        thick,
        font=\tiny,
        align=center
    }
]

% ---- Infrastructure substrate the system runs on ----
